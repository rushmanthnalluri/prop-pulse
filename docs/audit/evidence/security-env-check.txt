# Directory containing the dataset tree (data/{raw,interim,processed,external}).

API_HOST=127.0.0.1

# Frontend → backend base URL (Vite).
VITE_API_URL=http://localhost:8000

# Logging verbosity: DEBUG | INFO | WARNING | ERROR.
LOG_LEVEL=INFO

# JSONL file where the backend logs every prediction (best-effort, rotating-safe).
PREDICTION_LOG_PATH=logs/predictions.jsonl

# PSI threshold at which a feature is considered drifted (warn = half of this).

# (Vite dev server :5173, docker-compose frontend :8080).
