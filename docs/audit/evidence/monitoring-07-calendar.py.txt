"""Task 7 — YrSold/sale_year structural drift caveat.

1. Exact PSI of an all-2026 YrSold/sale_year production sample vs the train
   reference (train YrSold <= 2008 by the time-based split, ADR-4).
2. End-to-end: n=250 synthetic 'today-dated' (2026-08-07) serving rows whose
   non-calendar features are resampled from train (seed 42) and whose
   calendar-derived features are remapped to today -> does drift_detected /
   retraining_recommended flip PURELY from calendar features?
3. Docs disclosure inventory.
"""
import json
import tempfile
from pathlib import Path

import numpy as np
import pandas as pd

from ml.monitoring.psi import bin_proportions, population_stability_index
from ml.monitoring.drift_check import run_drift_check

ref = json.load(open("models/monitoring/reference_stats.json"))

print("=== A. Direct PSI: all-2026 production vs train reference ===")
for feat in ["YrSold", "sale_year"]:
    spec = ref["numeric"][feat]
    props = bin_proportions([2026.0] * 500, spec["bin_edges"])
    psi = population_stability_index(spec["expected_proportions"], props)
    print(f"{feat}: edges={spec['bin_edges']} expected={[round(p,4) for p in spec['expected_proportions']]}")
    print(f"  props(all-2026)={props.tolist()} PSI={psi:.6f}  (live report shows 4.358638)")
print("-> any post-2008 production value lands in the open last bin; PSI ~= 4.36 >> 0.2 forever")

print("\n=== B. Split year ranges (time-based split, ADR-4) ===")
for split in ["train", "val", "test"]:
    df = pd.read_csv(f"data/processed/{split}.csv", keep_default_na=False)
    ys = pd.to_numeric(df["YrSold"], errors="coerce")
    print(f"  {split}: YrSold {ys.min()}–{ys.max()}, n={len(df)}")

print("\n=== C. Full drift check on n=250 synthetic today-dated serving rows ===")
rng = np.random.default_rng(42)
from ml.features.pipeline import build_feature_frame
from ml.features.stats import fit_neighborhood_stats

raw_train = pd.read_csv("data/processed/train.csv", keep_default_na=False)
train = build_feature_frame(raw_train, fit_neighborhood_stats(raw_train))
num_feats = ref["numeric_features"]
base = {}
for f in num_feats:
    col = pd.to_numeric(train[f], errors="coerce").dropna().to_numpy()
    base[f] = rng.choice(col, size=250, replace=True) if len(col) else np.zeros(250)

TODAY_YEAR, TODAY_MONTH = 2026, 8
TODAY_QUARTER = (TODAY_MONTH - 1) // 3 + 1
CALENDAR_OVERRIDES = {
    "YrSold": TODAY_YEAR, "sale_year": TODAY_YEAR,
    "MoSold": TODAY_MONTH, "sale_month": TODAY_MONTH,
    "sale_quarter": TODAY_QUARTER,
}
# property_age / years_since_remod move with the calendar too: same housing
# stock, +18 years vs the latest train year (2026-2008).
ADD18 = {"property_age": 18, "years_since_remod": 18}

with tempfile.TemporaryDirectory() as tmp:
    log = Path(tmp) / "today.jsonl"
    with log.open("w", encoding="utf-8") as fh:
        for i in range(250):
            feats = {}
            for f in num_feats:
                v = float(base[f][i])
                if f in CALENDAR_OVERRIDES:
                    v = float(CALENDAR_OVERRIDES[f])
                elif f in ADD18:
                    v = v + ADD18[f]
                feats[f] = v
            rec = {"timestamp": "2026-08-07T12:00:00+00:00", "payload": {},
                   "features": feats,
                   "prediction": {"estimated_price": 200000.0, "probability": 0.5, "cluster_id": 0},
                   "model_version": "t"}
            fh.write(json.dumps(rec) + "\n")
    rep = run_drift_check(log_path=log, output_path=Path(tmp) / "report.json")
    print(f"n={rep['n_predictions']} drift_detected={rep['drift_detected']} retraining_recommended={rep['retraining_recommended']}")
    print(f"drifted_features={rep['drifted_features']}")
    print(f"warn_features={rep['warn_features']}")
    cal = {"YrSold", "sale_year", "MoSold", "sale_month", "sale_quarter", "property_age", "years_since_remod"}
    non_cal_drifted = [f for f in rep["drifted_features"] if f not in cal]
    print(f"drifted NON-calendar features: {non_cal_drifted}")
    for f in rep["drifted_features"]:
        print(f"  {f}: PSI={rep['per_feature_psi'][f]}")
    print(f"recommendation_text: {rep['recommendation_text'][:200]}")

print("\n=== D. Docs disclosure inventory ===")
print("docs/API.md:117-119 — 'drifted_features: [YrSold, sale_year] is expected on any post-2008 traffic ... Treat pure calendar drift as structural.'")
print("README.md:379-381 — 'Known structural caveat: YrSold/sale_year always \"drift\" on post-2008 traffic ... ignore pure calendar features'")
print("docs/DEPLOYMENT.md:132-135 — 'Expect YrSold/sale_year to always flag on post-2008 traffic (structural — the train split ends at 2008)'")
print("reports/DOCKER_SMOKE.md:189 — mentions known YrSold/sale_year caveat")
print("\nALL TASK-7 CHECKS DONE")
