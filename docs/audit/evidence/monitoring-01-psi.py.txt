"""Task 1 — independent PSI correctness verification (monitoring auditor).

Recomputes PSI WITHOUT using ml.monitoring.psi for the "independent" side:
own binning ([-inf] + interior edges + [+inf]) and own Σ(a−e)·ln(a/e) with
manual eps handling; compares against the library implementation.
"""
import json
import math
import numpy as np

from ml.monitoring.psi import (
    bin_proportions,
    population_stability_index,
    psi_bins_from_train,
)

ref = json.load(open("models/monitoring/reference_stats.json"))


def independent_proportions(values, edges):
    """Hand-built binning: outer edges open (±inf)."""
    bounds = np.asarray(edges, dtype=float)
    full = np.concatenate(([-np.inf], bounds[1:-1], [np.inf]))
    vals = np.asarray([float(v) for v in values], dtype=float)
    vals = vals[np.isfinite(vals)]
    counts, _ = np.histogram(vals, bins=full)
    return counts / counts.sum()


def independent_psi(expected, actual, eps=1e-6):
    """Hand-built PSI: normalize, clip at eps, renormalize, Σ(a−e)·ln(a/e)."""
    e = np.asarray(expected, dtype=float)
    a = np.asarray(actual, dtype=float)
    e = e / e.sum()
    a = a / a.sum()
    e = np.clip(e, eps, None); e /= e.sum()
    a = np.clip(a, eps, None); a /= a.sum()
    return float(np.sum((a - e) * np.log(a / e)))


print("=== A. Hand-computed formula check (no code but mine) ===")
# e=[0.5,0.5], a=[0.25,0.75]: PSI = (0.25-0.5)*ln(0.5) + (0.75-0.5)*ln(1.5)
hand = (0.25 - 0.5) * math.log(0.25 / 0.5) + (0.75 - 0.5) * math.log(0.75 / 0.5)
lib = population_stability_index([0.5, 0.5], [0.25, 0.75])
print(f"hand={hand!r}  lib={lib!r}  match={abs(hand-lib) < 1e-12}")

print("\n=== B. Three reference features vs hand-built shifted samples (seed 42) ===")
rng = np.random.default_rng(42)
FEATURES = ["GrLivArea", "OverallQual", "LotArea"]
for feat in FEATURES:
    spec = ref["numeric"][feat]
    edges = spec["bin_edges"]
    expected = spec["expected_proportions"]
    print(f"\n-- {feat}: {len(edges)-1} bins, edges={[round(e,2) for e in edges]}")
    print(f"   expected proportions={[round(p,4) for p in expected]}")

    # Hand-built shifted sample: normal around a point shifted +25% above the
    # train median (approximated by the middle edge region), 500 draws.
    median_approx = edges[len(edges) // 2]
    spread = max((edges[-1] - edges[0]) / 6.0, 1.0)
    sample = rng.normal(loc=median_approx * 1.25, scale=spread, size=500)
    sample = np.clip(sample, 0.0, None)  # areas/quality are non-negative

    lib_props = bin_proportions(sample, edges)
    ind_props = independent_proportions(sample, edges)
    lib_psi = population_stability_index(expected, lib_props)
    ind_psi = independent_psi(expected, ind_props)
    print(f"   lib  props={[round(p,4) for p in lib_props]}")
    print(f"   indep props={[round(p,4) for p in ind_props]}")
    print(f"   PSI lib={lib_psi:.10f}  indep={ind_psi:.10f}  match={abs(lib_psi-ind_psi)<1e-9}")

    # In-distribution control: sample built to follow expected proportions.
    draws = []
    for i, p in enumerate(expected):
        lo = -1e9 if i == 0 else edges[i]
        hi = 1e9 if i == len(edges) - 1 else edges[i + 1]
        lo = max(lo, edges[0]); hi = min(hi, edges[-1])
        draws.append(rng.uniform(lo, hi, size=max(1, int(round(p * 500)))))
    control = np.concatenate(draws)
    psi_control = population_stability_index(expected, bin_proportions(control, edges))
    print(f"   in-distribution control PSI={psi_control:.6f} (should be small)")

print("\n=== C. eps handling — empty-bin penalty ===")
# expected has 5 equal bins; actual puts 20% mass into bin 0 and 80% into bin 4,
# leaving bins 1..3 empty -> clip at 1e-6 then renormalize.
expected = [0.2] * 5
actual = [0.2, 0.0, 0.0, 0.0, 0.8]
lib = population_stability_index(expected, actual)
# hand: a' = clip(a,1e-6)/sum ; term per empty bin = (eps'−0.2)*ln(eps'/0.2)
a = np.clip(np.asarray(actual) / 1.0, 1e-6, None); a /= a.sum()
hand = float(np.sum((a - 0.2) * np.log(a / 0.2)))
print(f"lib={lib:.10f} hand={hand:.10f} match={abs(lib-hand)<1e-9}")
print(f"finite={math.isfinite(lib)}; penalty vs identical={lib - population_stability_index(expected, expected):.3f}")
# degenerate case: actual entirely in a bin where expected ~0 (clip applies both sides)
lib2 = population_stability_index([1e-9, 1.0], [1.0, 0.0])
print(f"surprise-bin PSI (1e-9 expected vs full mass) = {lib2:.4f} (large but finite: {math.isfinite(lib2)})")

print("\n=== D. Out-of-range values land in edge bins (±inf open edges) ===")
edges = ref["numeric"]["GrLivArea"]["bin_edges"]
lo, hi = edges[0], edges[-1]
print(f"GrLivArea train range edges: min={lo}, max={hi}")
vals = [lo - 5000.0, hi + 50000.0]  # far outside train range
props = bin_proportions(vals, edges)
print(f"props for [min-5000, max+50000] = {props.tolist()}")
print(f"  -> first bin={props[0]}, last bin={props[-1]}, interior sum={props[1:-1].sum()}")
assert props[0] == 0.5 and props[-1] == 0.5 and props[1:-1].sum() == 0.0
# and PSI actually inflates for a fully out-of-range production sample
props_allhigh = bin_proportions([hi * 10.0] * 100, edges)
psi_oob = population_stability_index(ref["numeric"]["GrLivArea"]["expected_proportions"], props_allhigh)
print(f"PSI of all-above-max production sample vs train ref = {psi_oob:.4f} (drift-scale)")

print("\n=== E. psi_bins_from_train conventions ===")
# constant sample -> [c-0.5, c, c+0.5]
print("constant sample edges:", psi_bins_from_train([7.0] * 100))
# zero-inflated -> duplicate quantiles deduped
zi = [0.0] * 95 + list(rng.uniform(1, 500, size=5))
print("zero-inflated (95% zero) edges:", psi_bins_from_train(zi))
# error paths
for bad in ([], [float("nan"), "x"]):
    try:
        psi_bins_from_train(bad)
        print(f"ERROR: {bad!r} did not raise")
    except ValueError as exc:
        print(f"empty/junk sample raises ValueError: {exc}")
try:
    psi_bins_from_train([1.0, 2.0], n_bins=0)
except ValueError as exc:
    print(f"n_bins=0 raises ValueError: {exc}")
print("\nALL TASK-1 CHECKS DONE")
