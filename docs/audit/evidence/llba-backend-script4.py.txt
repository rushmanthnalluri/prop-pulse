MIN = {"neighborhood": "NAmes", "bedrooms": 3, "full_bath": 2, "half_bath": 1,

# --- 1. lifespan fail-fast: champion.json present but artifact paths bogus ---

    "regression": {"name": "r", "version": "v0", "path": "models/registry/NOPE.joblib",
                   "residual_interval": {"q_low": -0.1, "q_high": 0.1}},
    "classification": {"name": "c", "version": "v0", "path": "models/registry/NOPE2.joblib",
                       "threshold": 0.5},
}))
try:
    bad_app = create_app(Settings(model_dir=str(fake_dir),
                                  prediction_log_path=str(tmp / "x.jsonl")))
    with TestClient(bad_app):
        print("lifespan fail-fast: DID NOT RAISE (bad)")

    print("lifespan fail-fast: RuntimeError:", exc)

    empty_app = create_app(Settings(model_dir=str(tmp / "empty"),
