print("\n--- JSONL writer thread-safety ---")
mt_log = tmp / "mt.jsonl"
logger = PredictionLogger(mt_log)
errors = []

def writer(tag):
    try:
        for i in range(25):
            logger.log(payload={"tag": tag, "i": i}, features={"f": i},
                       prediction={"estimated_price": 1.0, "probability": None, "cluster_id": 0},
                       model_version="t")
    except Exception as exc:  # noqa: BLE001
        errors.append(exc)

threads = [threading.Thread(target=writer, args=(t,)) for t in range(8)]
for t in threads: t.start()
for t in threads: t.join()
lines = mt_log.read_text("utf-8").splitlines()
parsed = [json.loads(line) for line in lines]
print("threads: 8 x 25 writes | exceptions:", len(errors),
      "| lines:", len(lines), "| all valid JSON:", len(parsed) == 200,
      "| distinct tags:", len({p['payload']['tag'] for p in parsed}))
